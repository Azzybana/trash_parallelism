1/// System utilities for time handling, environment variables, file system operations, and system interactions.
2///
3/// This module provides convenient wrappers for system-level operations including
4/// time manipulation, environment variable access, path operations, and file metadata.
5///
6/// ## Submodules
7///
8/// - [`datetime`]: Date and time utilities with serialization and parallel processing
9/// - [`env`]: Environment variable handling with JSON parsing and validation
10/// - [`path`]: File system operations with parallel and async capabilities
11///
12/// ## Backward Compatibility
13///
14/// All functions are re-exported at the module level for backward compatibility.
15/// You can access them directly as `trash_analyzer::sys::function_name()` or
16/// use the submodules for more organized access.
17// Standard library imports
18use std::time::{Duration, Instant};
19
20// Submodule declarations
21pub mod datetime;
22pub mod env;
23pub mod path;
24
25// Re-exports for backward compatibility

30/// Performance timer
31pub struct Timer {
32    start: Instant,
33    label: String,
34}
35
36impl Timer {
37    /// Start a new timer
38    #[must_use]
39    pub fn new(label: &str) -> Self {
40        Self {
41            start: Instant::now(),
42            label: label.to_string(),
43        }
44    }
45
46    /// Get elapsed time
47    #[must_use]
48    pub fn elapsed(&self) -> Duration {
49        self.start.elapsed()
50    }
51}
52
53impl Drop for Timer {
54    fn drop(&mut self) {
55        let duration = self.start.elapsed();
56        println!("Timer '{}' completed in {:?}", self.label, duration);
57    }
58}
