28/// Parallel string processing: split string and process chunks
29pub fn parallel_process_string<F, R>(s: &str, chunk_size: usize, processor: F) -> Vec<R>
30where
31    F: Fn(&str) -> R + Send + Sync,
32    R: Send,
33{
34    let chunks: Vec<&str> = s
35        .as_bytes()
36        .chunks(chunk_size)
37        .map(|chunk| std::str::from_utf8(chunk).unwrap_or(""))
38        .collect();
39
40    chunks.into_iter().map(processor).collect()
41}

43/// Asynchronously read a file to string
44///
45/// # Errors
46///
47/// Returns an `std::io::Error` if the file cannot be opened or read.
48pub async fn read_file_to_string_async(path: &str) -> Result<String, std::io::Error> {
49    let mut file = fs::File::open(path).await?;
50    let mut contents = String::new();
51    file.read_to_string(&mut contents).await?;
52    Ok(contents)
53}
54
55/// Asynchronously write string to file
56///
57/// # Errors
58///
59/// Returns an `std::io::Error` if the file cannot be created or written to.
60pub async fn write_string_to_file_async(path: &str, contents: &str) -> Result<(), std::io::Error> {
61    let mut file = fs::File::create(path).await?;
62    file.write_all(contents.as_bytes()).await?;
63    file.flush().await?;
64    Ok(())
65}

67/// Extract all JSON values by key from a JSON array string
68///
69/// # Errors
70///
71/// Returns an error if the input is not valid JSON or if parsing fails.
72pub fn extract_json_values_by_key(
73    json_array: &str,
74    key: &str,
75) -> Result<Vec<serde_json::Value>, Box<dyn std::error::Error>> {
76    let array: Vec<serde_json::Value> = serde_json::from_str(json_array)?;
77    let mut results = Vec::new();
78
79    for item in array {
80        if let Some(obj) = item.as_object()
81            && let Some(value) = obj.get(key)
82        {
83            results.push(value.clone());
84        }
85    }
86
87    Ok(results)
88}
