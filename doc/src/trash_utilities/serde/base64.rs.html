1/// Encode data as base64 string.
2///
3/// This function encodes binary data to a base64 string using standard encoding.
4/// Useful for transmitting binary data over text-based protocols.
5///
6/// # Parameters
7/// - `data`: The binary data to encode.
8///
9/// # Returns
10/// A base64-encoded string.
11///
12/// # Examples
13/// ```rust
14/// use trash_analyzer::serde::encode_base64;
15///
16/// let data = b"Hello, world!";
17/// let encoded = encode_base64(data);
18/// println!("Base64: {}", encoded);
19/// ```
20#[must_use]
21pub fn encode_base64(data: &[u8]) -> String {
22    crate::common::encode_base64(data)
23}

25/// Decode a base64 string to binary data.
26///
27/// This function decodes a base64 string back to binary data.
28/// Returns an error if the input is not valid base64.
29///
30/// # Parameters
31/// - `data`: The base64 string to decode.
32///
33/// # Returns
34/// - `Ok(Vec<u8>)` containing the decoded bytes.
35/// - `Err(base64::DecodeError)` if decoding fails.
36///
37/// # Errors
38/// Returns `base64::DecodeError` if the input string is not valid base64.
39///
40/// # Examples
41/// ```rust
42/// use trash_analyzer::serde::{encode_base64, decode_base64};
43///
44/// let data = b"Hello, world!";
45/// let encoded = encode_base64(data);
46/// let decoded = decode_base64(&encoded).unwrap();
47/// assert_eq!(decoded, data);
48/// ```
49pub fn decode_base64(data: &str) -> Result<Vec<u8>, base64::DecodeError> {
50    crate::common::decode_base64(data)
51}

53/// Serialize a value to JSON and encode it as base64.
54///
55/// This convenience function serializes a value to JSON bytes and then
56/// encodes those bytes as base64. Useful for embedding structured data
57/// in text formats like JSON or XML.
58///
59/// # Type Parameters
60/// - `T`: The type to serialize, must implement `Serialize`.
61///
62/// # Parameters
63/// - `value`: The value to serialize and encode.
64///
65/// # Returns
66/// - `Ok(String)` containing the base64-encoded JSON.
67/// - `Err(serde_json::Error)` if serialization fails.
68///
69/// # Errors
70/// Returns `serde_json::Error` if the value cannot be serialized to JSON.
71///
72/// # Examples
73/// ```rust
74/// use trash_analyzer::serde::serialize_to_base64_json;
75/// use serde::Serialize;
76///
77/// #[derive(Serialize)]
78/// struct Person { name: String, age: u32 }
79///
80/// let person = Person { name: "Alice".to_string(), age: 30 };
81/// let b64_json = serialize_to_base64_json(&person).unwrap();
82/// println!("Base64 JSON: {}", b64_json);
83/// ```
84pub fn serialize_to_base64_json<T: serde::Serialize>(
85    value: &T,
86) -> Result<String, serde_json::Error> {
87    let bytes = serde_json::to_vec(value)?;
88    Ok(encode_base64(&bytes))
89}

91/// Decode base64 JSON and deserialize to a struct.
92///
93/// This function decodes a base64 string to JSON bytes and then deserializes
94/// those bytes into a struct. The inverse of `serialize_to_base64_json`.
95///
96/// # Type Parameters
97/// - `T`: The type to deserialize into, must implement `DeserializeOwned`.
98///
99/// # Parameters
100/// - `b64_json`: The base64-encoded JSON string.
101///
102/// # Returns
103/// - `Ok(T)` containing the deserialized value.
104/// - `Err(Box<dyn std::error::Error>)` if decoding or deserialization fails.
105///
106/// # Errors
107/// Returns an error if the base64 string is invalid or if the decoded JSON cannot be deserialized into the target type.
108///
109/// # Examples
110/// ```rust
111/// use trash_analyzer::serde::{serialize_to_base64_json, deserialize_from_base64_json};
112/// use serde::{Serialize, Deserialize};
113///
114/// #[derive(Serialize, Deserialize, Debug, PartialEq)]
115/// struct Person { name: String, age: u32 }
116///
117/// let person = Person { name: "Alice".to_string(), age: 30 };
118/// let b64_json = serialize_to_base64_json(&person).unwrap();
119/// let decoded: Person = deserialize_from_base64_json(&b64_json).unwrap();
120/// assert_eq!(decoded, person);
121/// ```
122pub fn deserialize_from_base64_json<T: for<'de> serde::Deserialize<'de>>(
123    b64_json: &str,
124) -> Result<T, Box<dyn std::error::Error>> {
125    let bytes = decode_base64(b64_json)?;
126    let value: T = serde_json::from_slice(&bytes)?;
127    Ok(value)
128}
