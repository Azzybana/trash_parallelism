13/// Parses key-value pairs from a string content, typically from config files.
14///
15/// This function assumes each line is in the format "key=value". It uses `memchr`
16/// for efficient byte-level searching of the '=' character. Lines without '=' are ignored.
17///
18/// # Parameters
19/// - `content`: The string content to parse.
20///
21/// # Returns
22/// An `AHashMap<String, String>` containing the parsed key-value pairs.
23/// Keys and values are trimmed of whitespace.
24///
25/// # Examples
26/// ```rust
27/// use trash_analyzer::base::data::parse_key_value;
28/// use ahash::AHashMap;
29///
30/// let content = "name=John\nage=30\ncity=New York";
31/// let map = parse_key_value(content);
32/// assert_eq!(map.get("name"), Some(&"John".to_string()));
33/// assert_eq!(map.get("age"), Some(&"30".to_string()));
34/// ```
35#[must_use]
36pub fn parse_key_value(content: &str) -> AHashMap<String, String> {
37    let mut map = AHashMap::new();
38    for line in content.lines() {
39        if let Some(pos) = memchr(b'=', line.as_bytes()) {
40            let key = &line[..pos];
41            let value = &line[pos + 1..];
42            map.insert(key.to_string(), value.to_string());
43        }
44    }
45    map
46}
